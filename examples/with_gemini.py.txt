#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
Пример: Molecular AI + Google Gemini.
"""

import os
import sys

sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

from core.system import MolecularSystem
from adapters.factory import create_adapter


def main():
    print("=== Molecular AI + Gemini Demo ===")

    # 1. Симуляция
    sys = MolecularSystem(n_agents=3, dt=0.05, noise=0.02)
    sys.run(200)
    print(f"Sync after 200 steps: r = {sys.order_parameter():.3f}")

    # 2. Подключаем Gemini
    try:
        adapter = create_adapter("gemini", model="gemini-1.5-flash")
    except Exception as e:
        print(f"[!] {e}")
        return

    # 3. Отправляем задачу
    agent = sys.agents[0]
    orbital = sys.orbital.layers[0].orbital
    task = "Suggest one improvement for this multi-agent synchronization system."

    prompt = adapter.build_prompt(agent, orbital, task)
    print("\n--- PROMPT ---")
    print(prompt)
    print("--- END PROMPT ---\n")

    try:
        response = adapter.call_llm(prompt)
        print("--- GEMINI RESPONSE ---")
        print(response)
        print("--- END RESPONSE ---")
    except Exception as e:
        print(f"[!] LLM call failed: {e}")


if __name__ == "__main__":
    main()
